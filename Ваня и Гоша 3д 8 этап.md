func _physics_process(delta: float) -> void:
	var v3 := Vector3.ZERO
	var mul := 1.0

	match mode:
		Mode.CONTROLLED:
			v3 = Vector3(input_vector.x, 0.0, input_vector.y).limit_length(1.0)
		Mode.FOLLOW:
			if follow_target != null:
				var d: Vector3 = follow_target.global_position - global_position
				var dist: float = d.length()
				if dist > 5.0:
					if _lag_left > 0.0:
						_lag_left -= delta
					else:
						if enable_lag:
							_lag_wait -= delta
							if _lag_wait <= 0.0:
								_lag_wait = randf_range(3.5, 5.5)
								_lag_left = 0.9
						v3 = d.normalized()
						mul = 1.0 + clampf((dist - 3.0) / 2.0, 0.0, 0.4)
		Mode.AUTO:
			var d2: Vector3 = _auto_target - global_position
			if d2.length() <= 0.2:
				mode = Mode.IDLE
				arrived.emit()
			else:
				v3 = d2.normalized()
				mul = _auto_mul

	var pos_2d := Vector2(global_position.x, global_position.z)
	for z in slow_zones:
		if z.has_point(pos_2d):
			mul *= slow_mult

	var step: Vector3 = (v3 * speed * mul) + (extra_pull * delta)

	if mode == Mode.AUTO:
		var rest: Vector3 = _auto_target - global_position
		if step.length() > rest.length():
			step = rest

	_moving = step.length() > 0.01
	if _moving:
		_try_move(step)
		_walk_t += delta * 11.0
		var target_angle = atan2(v3.x, v3.z)
		rotation.y = lerp_angle(rotation.y, target_angle, 10.0 * delta)

	move_and_slide()

func _try_move(step: Vector3) -> void:
	var target: Vector3 = global_position + step
	if _ok(target):
		global_position = target
	elif _ok(Vector3(target.x, global_position.y, global_position.z)):
		global_position.x = target.x
	elif _ok(Vector3(global_position.x, global_position.y, target.z)):
		global_position.z = target.z

func _ok(p: Vector3) -> bool:
	if walkable.is_empty():
		return true
	var p_2d := Vector2(p.x, p.z)
	for r in walkable:
		if r.has_point(p_2d):
			return true
	return false

func walk_to(p: Vector3, mul: float = 1.0) -> void:
	_auto_target = p
	_auto_mul = mul
	mode = Mode.AUTO
	await arrived

func set_sitting(is_sitting: bool) -> void:
	sit = is_sitting
	if body_mesh and head_mesh:
		body_mesh.position.y = -0.4 if sit else 0.0
		head_mesh.position.y = 0.4 if sit else 0.8
