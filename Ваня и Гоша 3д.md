Да, конечно. Ниже собираю **пошаговый план переделки 2D → 3D** и к каждому шагу добавляю **что именно менять в коде**.

Важно: это пока **не финальный огромный код всех файлов сразу**, а понятная карта работы. Дальше мы можем идти **по одному этапу**: я присылаю полный файл → ты вставляешь → запускаешь → если ошибка, присылаешь консоль.

Главное правило: **старые 2D-файлы не затираем**. Делаем рядом новые:

```text
character_3d.gd

world_3d.gd
bird_3d.gd
critter_3d.gd
campfire_3d.gd
camp_world_3d.gd
owl_3d.gd

```

# Общий принцип переноса

В 2D у тебя мир был в пикселях:

```gdscript
Vector2(x, y)
```

В 3D будет так:

```gdscript
Vector3(x, y, z)
```

Где:

```text
2D x → 3D x
2D y → 3D z
3D y → высота
```

Чтобы не переписывать все координаты сценария вручную, вводим общий масштаб:

```text
100 пикселей 2D = 2 метра 3D
```

То есть:

```gdscript
const S := 0.02
```

# Шаг 0. Новый общий помощник координат

## Новый файл

```text
scripts/geo3d.gd
```

## Зачем

Чтобы не путаться в `Vector2` и `Vector3`.

## Код файла

```gdscript
class_name Geo3D
extends RefCounted

const S: float = 0.02

static func to3(p: Vector2, y: float = 0.0) -> Vector3:
	return Vector3(p.x * S, y, p.y * S)

static func to2(p: Vector3) -> Vector2:
	return Vector2(p.x / S, p.z / S)

static func dir3(v: Vector2) -> Vector3:
	return Vector3(v.x, 0.0, v.y)
```

## Что это даёт

Старые координаты сценария можно оставлять почти без изменений:

```gdscript
vanya.position = Geo3D.to3(Vector2(110, 305))
```

А проверки делать так:

```gdscript
Geo3D.to2(vanya.global_position).x > 260
```

## Движение

```gdscript
func _physics_process(delta: float) -> void:
	var v: Vector3 = Vector3.ZERO
	var mul: float = 1.0

	match mode:
		Mode.CONTROLLED:
			v = input_vector.limit_length(1.0)

		Mode.FOLLOW:
			if follow_target != null:
				var d: Vector3 = follow_target.global_position - global_position
				d.y = 0.0
				var dist: float = d.length()

				if dist > 5.2:
					var warp: Vector3 = follow_target.global_position - d.normalized() * follow_distance
					global_position = warp if _ok(warp) else follow_target.global_position
					d = follow_target.global_position - global_position
					dist = d.length()

				if _lag_left > 0.0:
					_lag_left -= delta
				else:
					if enable_lag:
						_lag_wait -= delta
						if _lag_wait <= 0.0:
							_lag_wait = randf_range(3.5, 5.5)
							_lag_left = 0.9

					if dist > follow_distance:
						v = d.normalized()
						mul = 1.0 + clampf((dist - follow_distance - 1.2) / 2.0, 0.0, 0.4)

		Mode.AUTO:
			var d2: Vector3 = _auto_target - global_position
			d2.y = 0.0

			if d2.length() <= 0.06:
				mode = Mode.IDLE
				arrived.emit()
			else:
				v = d2.normalized()
				mul = _auto_mul

	var p2: Vector2 = Geo3D.to2(global_position)

	for z in slow_zones:
		if z.has_point(p2):
			mul *= slow_mult

	var step: Vector3 = (v * speed * mul + extra_pull) * delta
	step.y = 0.0

	if mode == Mode.AUTO:
		var rest: Vector3 = _auto_target - global_position
		rest.y = 0.0
		if step.length() > rest.length():
			step = rest

	_moving = step.length() > 0.001

	if _moving:
		_try_move(step)
		_walk_t += delta * 11.0

		var flat_v: Vector3 = v
		flat_v.y = 0.0
		if flat_v.length() > 0.05:
			var target_angle: float = atan2(flat_v.x, flat_v.z)
			rotation.y = lerp_angle(rotation.y, target_angle, 10.0 * delta)
```

func _make_hand_link() -> void:
	hand_link = MeshInstance3D.new()
	add_child(hand_link)


func _update_hand_link() -> void:
	if hand_link == null:
		return

	if not holding:
		hand_link.visible = false
		return

	var a: Vector3 = vanya.global_position + Vector3(0, 0.75, 0)
	var b: Vector3 = gosha.global_position + Vector3(0, 0.65, 0)

		hand_link.visible = false
		return

	mesh.bottom_radius = 0.025

	hand_link.global_position = mid

	hand_link.rotate_object_local(Vector3.RIGHT, PI * 0.5)
