## 🐦 Этап 5: Синяя птичка в 3D (`bird_3d.gd`)

### 📝 Создай файл `scripts/bird_3d.gd`:

```gdscript
extends Node3D

# Синяя птичка. Появляется, подлетает к героям, улетает.
# Визуал: простые 3D-примитивы (сферы, боксы), чтобы не искать модели.

var _body: MeshInstance3D
var _wing_l: MeshInstance3D
var _wing_r: MeshInstance3D
var _beak: MeshInstance3D
var _eye: MeshInstance3D

var _t: float = 0.0
var _flying: bool = false
var _fly_from: Vector3 = Vector3.ZERO
var _fly_to: Vector3 = Vector3.ZERO
var _fly_dur: float = 1.0
var _fly_elapsed: float = 0.0
var _hover_offset: Vector3 = Vector3.ZERO

func _ready() -> void:
	visible = false
	_build_mesh()

func _build_mesh() -> void:
	# Тело (синяя сфера)
	_body = MeshInstance3D.new()
	var body_mesh = SphereMesh.new()
	body_mesh.radius = 0.25
	body_mesh.height = 0.5
	_body.mesh = body_mesh
	var body_mat = StandardMaterial3D.new()
	body_mat.albedo_color = Color(0.55, 0.78, 1.0)
	_body.material_override = body_mat
	_body.position.y = 0.3
	add_child(_body)
	
	# Голова (чуть меньше, спереди)
	var head = MeshInstance3D.new()
	var head_mesh = SphereMesh.new()
	head_mesh.radius = 0.18
	head_mesh.height = 0.36
	head.mesh = head_mesh
	head.material_override = body_mat
	head.position = Vector3(0.2, 0.45, 0)
	_body.add_child(head)
	
	# Клюв (маленький конус)
	_beak = MeshInstance3D.new()
	var beak_mesh = CylinderMesh.new()
	beak_mesh.top_radius = 0.0
	beak_mesh.bottom_radius = 0.05
	beak_mesh.height = 0.12
	_beak.mesh = beak_mesh
	var beak_mat = StandardMaterial3D.new()
	beak_mat.albedo_color = Color(1.0, 0.7, 0.2)
	_beak.material_override = beak_mat
	_beak.position = Vector3(0.38, 0.45, 0)
	_beak.rotation_degrees = Vector3(0, 0, -90)
	_body.add_child(_beak)
	
	# Глаз (чёрная точка)
	_eye = MeshInstance3D.new()
	var eye_mesh = SphereMesh.new()
	eye_mesh.radius = 0.03
	eye_mesh.height = 0.06
	_eye.mesh = eye_mesh
	var eye_mat = StandardMaterial3D.new()
	eye_mat.albedo_color = Color(0.05, 0.05, 0.08)
	_eye.material_override = eye_mat
	_eye.position = Vector3(0.32, 0.5, 0.1)
	_body.add_child(_eye)
	
	# Крылья (два плоских бокса)
	_wing_l = _make_wing()
	_wing_l.position = Vector3(-0.05, 0.35, 0.2)
	_body.add_child(_wing_l)
	
	_wing_r = _make_wing()
	_wing_r.position = Vector3(-0.05, 0.35, -0.2)
	_body.add_child(_wing_r)

func _make_wing() -> MeshInstance3D:
	var wing = MeshInstance3D.new()
	var wing_mesh = BoxMesh.new()
	wing_mesh.size = Vector3(0.3, 0.05, 0.18)
	wing.mesh = wing_mesh
	var wing_mat = StandardMaterial3D.new()
	wing_mat.albedo_color = Color(0.45, 0.68, 0.9)
	wing.material_override = wing_mat
	return wing

# ---------- Анимация ----------

func _process(delta: float) -> void:
	if not visible:
		return
	_t += delta
	
	# Полёт к цели
	if _flying:
		_fly_elapsed += delta
		var t = clampf(_fly_elapsed / _fly_dur, 0.0, 1.0)
		# Плавная кривая (дуга)
		var straight = _fly_from.lerp(_fly_to, t)
		var arc = Vector3(0, sin(t * PI) * 1.5, 0)  # подъём в середине пути
		global_position = straight + arc
		
		# Поворот в сторону движения
		var dir = (_fly_to - _fly_from).normalized()
		if dir.length() > 0.01:
			var target_angle = atan2(dir.x, dir.z)
			rotation.y = lerp_angle(rotation.y, target_angle, 8.0 * delta)
		
		if t >= 1.0:
			_flying = false
			global_position = _fly_to
		return
	
	# Парение на месте (лёгкое покачивание)
	_hover_offset.y = sin(_t * 2.5) * 0.08
	_body.position.y = 0.3 + _hover_offset.y
	
	# Взмахи крыльев
	var flap = sin(_t * 12.0) * 0.4
	_wing_l.rotation_degrees = Vector3(0, 0, flap * 30.0)
	_wing_r.rotation_degrees = Vector3(0, 0, -flap * 30.0)

# ---------- Публичные методы ----------

# Появиться в точке и подлететь к цели за dur секунд
func fly_to(target: Vector3, dur: float = 1.5) -> void:
	visible = true
	_fly_from = global_position
	_fly_to = target
	_fly_dur = dur
	_fly_elapsed = 0.0
	_flying = true
	
	# Ждём окончания полёта
	await get_tree().create_timer(dur).timeout

# Улететь вверх и скрыться
func leave() -> void:
	var up_target = global_position + Vector3(0, 8, -3)
	_fly_from = global_position
	_fly_to = up_target
	_fly_dur = 2.0
	_fly_elapsed = 0.0
	_flying = true
	
	await get_tree().create_timer(2.0).timeout
	visible = false
```

### 📝 Обновляем `scripts/main_3d.gd`

Теперь нужно добавить птичку в сценарий. Открой `scripts/main_3d.gd` и замени его **полностью** на этот код (я добавил птичку и сцену с ней, но пока **без ручья и камней** — это сделаем следующим шагом, чтобы не перегружать):

```gdscript
extends Node3D

const VanyaScene = preload("res://scenes/vanya.tscn")
const GoshaScene = preload("res://scenes/gosha.tscn")
const BirdScene = preload("res://scenes/bird.tscn")
const DialogueScript = preload("res://scripts/dialogue_box.gd")
const TouchScript = preload("res://scripts/touch_controls.gd")
const CritterScript = preload("res://scripts/critter_3d.gd")
const WorldScript = preload("res://scripts/world_3d.gd")

const NAMES = { "vanya": "Ваня", "gosha": "Гоша", "bird": "Птичка" }
const COLORS = {
	"vanya": Color(1.0, 0.5, 0.45),
	"gosha": Color(0.55, 0.9, 0.5),
	"bird": Color(0.55, 0.78, 1.0),
}

enum State {
	INTRO_WALK,
	BIRD_SCENE,
	REFUSAL_WALK,
	END,
}

var state: int = State.INTRO_WALK
var world
var vanya: CharacterBody3D
var gosha: CharacterBody3D
var bird
var box: CanvasLayer
var touch: Control
var camera: Camera3D
var dim: CanvasModulate

var controlled = null
var holding: bool = true
var _locked: bool = false

const CAM_OFFSET := Vector3(0, 10, 12)
const CAM_LOOK_AT_OFFSET := Vector3(0, 0.5, 0)

# Лесные жители (3D координаты)
const CRITTERS_3D = [
	["squirrel", Vector3(15, 0, 3)],
	["rabbit", Vector3(8, 0, 12)],
	["hedgehog", Vector3(17, 0, 11)],
	["deer", Vector3(22, 0, 1)],
	["frog", Vector3(25, 0, 15)],
	["squirrel", Vector3(33, 0, 14)],
]

func _ready() -> void:
	# 1. Тёплый летний свет
	dim = CanvasModulate.new()
	dim.color = Color(1.0, 0.97, 0.88)
	add_child(dim)

	# 2. Перспективная камера под углом
	camera = Camera3D.new()
	camera.projection = Camera3D.PROJECTION_PERSPECTIVE
	camera.fov = 55.0
	camera.position = Vector3(0, 10, 12)
	camera.rotation_degrees = Vector3(-40, 0, 0)
	add_child(camera)
	camera.make_current()

	# 3. Лес
	world = WorldScript.new()
	add_child(world)

	# 4. Персонажи
	vanya = VanyaScene.instantiate()
	vanya.position = Vector3(0, 0, 0)
	add_child(vanya)

	gosha = GoshaScene.instantiate()
	gosha.position = Vector3(-1.5, 0, 0.5)
	gosha.enable_lag = true
	add_child(gosha)

	# 5. Птичка
	bird = BirdScene.instantiate()
	add_child(bird)

	# 6. Настройка walkable зон
	var walk: Array[Rect2] = world.walkable_start()
	var slow: Array[Rect2] = world.slow_zones()
	vanya.walkable = walk
	vanya.slow_zones = slow
	gosha.walkable = walk
	gosha.slow_zones = slow

	# 7. Лесные жители
	var water: Array[Rect2] = [
		Rect2(26, 16, 2.5, 4.5),
		Rect2(42, 16, 8.0, 5.0),
	]
	var no_zones: Array[Rect2] = []
	for spec in CRITTERS_3D:
		var cr = CritterScript.new()
		cr.kind = spec[0]
		cr.global_position = spec[1]
		cr.home = spec[1]
		cr.heroes = [vanya, gosha]
		cr.avoid = no_zones if spec[0] == "frog" else water
		add_child(cr)

	# 8. UI
	box = DialogueScript.new()
	add_child(box)
	
	touch = TouchScript.new()
	add_child(touch)

	_set_control(vanya)
	
	await get_tree().create_timer(0.5).timeout
	_run_act()

func _process(delta: float) -> void:
	if controlled != null and not _locked:
		var input_2d = touch.vector
		if Input.is_action_pressed("ui_right"): input_2d.x += 1.0
		if Input.is_action_pressed("ui_left"): input_2d.x -= 1.0
		if Input.is_action_pressed("ui_down"): input_2d.y += 1.0
		if Input.is_action_pressed("ui_up"): input_2d.y -= 1.0
		
		if input_2d.length() > 0.1:
			var cam_right = camera.global_transform.basis.x
			cam_right.y = 0
			cam_right = cam_right.normalized()
			
			var cam_forward = -camera.global_transform.basis.z
			cam_forward.y = 0
			cam_forward = cam_forward.normalized()
			
			var move_dir = cam_right * input_2d.x - cam_forward * input_2d.y
			controlled.input_vector = move_dir.limit_length(1.0)
		else:
			controlled.input_vector = Vector2.ZERO

func _physics_process(delta: float) -> void:
	if controlled != null:
		var target_pos = controlled.global_position + CAM_OFFSET
		camera.global_position = camera.global_position.lerp(target_pos, 1.0 - exp(-5.0 * delta))
		
		var look_target = controlled.global_position + CAM_LOOK_AT_OFFSET
		camera.look_at(look_target, Vector3.UP)

# ---------- Управление ----------
func _set_control(ch) -> void:
	controlled = ch
	var other = gosha if ch == vanya else vanya
	ch.mode = ch.Mode.CONTROLLED
	other.mode = other.Mode.FOLLOW

func _lock() -> void:
	_locked = true
	touch.enabled = false
	if vanya: vanya.mode = vanya.Mode.IDLE
	if gosha: gosha.mode = gosha.Mode.IDLE
	if controlled: controlled.input_vector = Vector2.ZERO

func _unlock(ch = null) -> void:
	if ch != null:
		controlled = ch
	_locked = false
	touch.enabled = true
	_set_control(controlled)

func _say(who: String, text: String) -> void:
	await box.say(NAMES[who], text, COLORS[who])

func _talk(lines: Array) -> void:
	_lock()
	for l in lines:
		await _say(l[0], l[1])
	box.hide_box()
	_unlock()

func _wait_until(cond: Callable) -> void:
	while not cond.call():
		await get_tree().process_frame

# ---------- Сценарий акта (с птичкой) ----------
func _run_act() -> void:
	state = State.INTRO_WALK
	box.toast("Веди Ваню: води пальцем по левой половине экрана", Color.WHITE, 3.5)
	
	await _wait_until(func(): return vanya.global_position.z < -8.0)
	await _talk([["vanya", "Не отставай. До дома недалеко."]])
	
	await _wait_until(func(): return vanya.global_position.z < -16.0)
	await _talk([
		["gosha", "Там белка была…"],
		["vanya", "Потом. Сейчас главное — не задерживаться."],
	])

	# ВСТРЕЧА С СИНЕЙ ПТИЧКОЙ
	state = State.BIRD_SCENE
	await _wait_until(func(): return vanya.global_position.z < -24.0)
	_lock()
	gosha.enable_lag = false
	
	# Птичка появляется справа и подлетает к Ване
	bird.global_position = Vector3(vanya.global_position.x + 6, 2.5, vanya.global_position.z - 2)
	await bird.fly_to(Vector3(vanya.global_position.x + 1.5, 1.2, vanya.global_position.z), 1.6)
	
	await _say("bird", "Подож… подождите.")
	await _say("bird", "Деду Морозу… очень плохо.")
	await _say("bird", "Правда плохо. Лесу нужна… ваша помощь.")
	await _say("bird", "Вы же… дети. Дети иногда могут то, что взрослые… не могут.")
	await _say("gosha", "Правда? А что случилось?")
	box.hide_box()
	
	# Первое принудительное переключение: 6 секунд за Гошу
	await _tug_scene()
	
	await _say("vanya", "Мы идём домой.")
	await _say("vanya", "Нас не просили. И мы никому ничего не должны.")
	await _say("vanya", "Я отвечаю за нас двоих. Мы не будем ввязываться в чужие дела.")
	box.hide_box()
	
	# Птичка улетает
	bird.leave()
	
	_unlock(vanya)
	
	state = State.END
	await box.end_card("Конец Первого акта (3D Тест)", "Нажми, чтобы продолжить")
	get_tree().change_scene_to_file("res://scenes/menu.tscn")

# Гоша тянется к птичке, Ваня оттягивает его за руку (6 секунд)
func _tug_scene() -> void:
	box.flash()
	box.toast("Теперь ты — Гоша", COLORS["gosha"], 2.0)
	_locked = false
	touch.enabled = true
	controlled = gosha
	gosha.mode = gosha.Mode.CONTROLLED
	vanya.mode = vanya.Mode.IDLE
	
	var t: float = 0.0
	while t < 6.0:
		t += get_process_delta_time()
		var to_v: Vector3 = vanya.global_position - gosha.global_position
		var pull: float = 3.0 * minf(t / 3.0, 1.0)  # сила притяжения
		if to_v.length() < 0.8:
			pull = 0.0
		gosha.extra_pull = to_v.normalized() * pull
		await get_tree().process_frame
	
	gosha.extra_pull = Vector3.ZERO
	_lock()
	box.flash()
```
