```gdscript
func _physics_process(delta: float) -> void:
	# Гравитация
	if not is_on_floor():
		velocity.y -= gravity * delta
	else:
		is_jumping = false
	
	var v3 := Vector3.ZERO
	var mul := 1.0

	match mode:
		Mode.CONTROLLED:
			v3 = Vector3(input_vector.x, 0.0, input_vector.y).limit_length(1.0)
		Mode.FOLLOW:
			if follow_target != null:
				# ПАРАЛЛЕЛЬНОЕ СЛЕДОВАНИЕ: Гоша идёт сбоку от Вани
				var target_pos = follow_target.global_position + follow_offset
				var d: Vector3 = target_pos - global_position
				var dist: float = d.length()
				
				if dist > 1.5:  # Начинаем двигаться, если далеко
					if _lag_left > 0.0:
						_lag_left -= delta
					else:
						if enable_lag:
							_lag_wait -= delta
							if _lag_wait <= 0.0:
								_lag_wait = randf_range(3.5, 5.5)
								_lag_left = 0.9
						v3 = d.normalized()
						mul = 1.0 + clampf((dist - 1.5) / 2.0, 0.0, 0.4)
		Mode.AUTO:
			var d2: Vector3 = _auto_target - global_position
			if d2.length() <= 0.2:
				mode = Mode.IDLE
				arrived.emit()
			else:
				v3 = d2.normalized()
				mul = _auto_mul

	var pos_2d := Vector2(global_position.x, global_position.z)
	for z in slow_zones:
		if z.has_point(pos_2d):
			mul *= slow_mult

	var step: Vector3 = (v3 * speed * mul) + (extra_pull * delta)

	if mode == Mode.AUTO:
		var rest: Vector3 = _auto_target - global_position
		if step.length() > rest.length():
			step = rest

	_moving = step.length() > 0.01
	if _moving:
		_try_move(step)
		_walk_t += delta * 11.0
		var target_angle = atan2(v3.x, v3.z)
		rotation.y = lerp_angle(rotation.y, target_angle, 10.0 * delta)

	# Применяем горизонтальную скорость
	velocity.x = step.x / delta if delta > 0 else 0
	velocity.z = step.z / delta if delta > 0 else 0
	
	move_and_slide()

func _try_move(step: Vector3) -> void:
	var target: Vector3 = global_position + step
	if _ok(target):
		global_position = target
	elif _ok(Vector3(target.x, global_position.y, target.z)):
		global_position.x = target.x
	elif _ok(Vector3(global_position.x, global_position.y, target.z)):
		global_position.z = target.z

func _ok(p: Vector3) -> bool:
	if walkable.is_empty():
		return true
	var p_2d := Vector2(p.x, p.z)
	for r in walkable:
		if r.has_point(p_2d):
			return true
	return false

func walk_to(p: Vector3, mul: float = 1.0) -> void:
	_auto_target = p
	_auto_mul = mul
	mode = Mode.AUTO
	await arrived

# ---------- ПРЫЖОК ----------
func jump() -> void:
	if is_on_floor() and not is_jumping:
		velocity.y = jump_velocity
		is_jumping = true

# ---------- ПРИСЕДАНИЕ ----------

	if body_mesh and head_mesh:
		body_mesh.position.y = -0.4 if sit else 0.0
		head_mesh.position.y = 0.4 if sit else 0.8
```

## 📷 Этап 3: Обновлённый `main_3d.gd` (орбитальная камера + кнопки)

Замени файл `scripts/main_3d.gd` **полностью** на этот код:

```gdscript

func _process(delta: float) -> void:
	if controlled != null and not _locked:
		var input_2d = touch.vector
		if Input.is_action_pressed("ui_right"): input_2d.x += 1.0
		if Input.is_action_pressed("ui_left"): input_2d.x -= 1.0
		if Input.is_action_pressed("ui_down"): input_2d.y += 1.0
		if Input.is_action_pressed("ui_up"): input_2d.y -= 1.0
		
		if input_2d.length() > 0.1:
			var cam_right = camera.global_transform.basis.x
			cam_right.y = 0
			cam_right = cam_right.normalized()
			
			var cam_forward = -camera.global_transform.basis.z
			cam_forward.y = 0
			cam_forward = cam_forward.normalized()
			
			var move_dir = cam_right * input_2d.x - cam_forward * input_2d.y
			controlled.input_vector = move_dir.limit_length(1.0)
		else:
			controlled.input_vector = Vector2.ZERO

func _physics_process(delta: float) -> void:
	if controlled != null:
		_update_camera_position()

# ---------- ОРБИТАЛЬНАЯ КАМЕРА ----------
func _update_camera_position() -> void:
	if controlled == null:
		return
	
	var distance = CAM_DISTANCE / zoom_level
	var pitch_rad = deg_to_rad(pitch)
	var yaw_rad = deg_to_rad(yaw)
	
	# Вычисляем позицию камеры на сфере вокруг персонажа
	var offset = Vector3(
		distance * cos(pitch_rad) * sin(yaw_rad),
		distance * sin(pitch_rad),
		distance * cos(pitch_rad) * cos(yaw_rad)
	)
	
	var target_pos = controlled.global_position + offset
	camera.global_position = camera.global_position.lerp(target_pos, 1.0 - exp(-5.0 * delta))
	
	var look_target = controlled.global_position + CAM_LOOK_AT_OFFSET
	camera.look_at(look_target, Vector3.UP)

# ---------- УПРАВЛЕНИЕ КАМЕРОЙ (свайпы по правой части экрана) ----------
func _input(event: InputEvent) -> void:
	if _locked:
		return
	
	# Свайп по правой половине экрана — вращение камеры
	if event is InputEventScreenTouch:
		if event.pressed and event.position.x > get_viewport().get_visible_rect().size.x * 0.65:
			_camera_touch_index = event.index
			_camera_last_pos = event.position
		elif event.index == _camera_touch_index:
			_camera_touch_index = -1
	
	elif event is InputEventScreenDrag and event.index == _camera_touch_index:
		var delta_pos = event.position - _camera_last_pos
		_camera_last_pos = event.position
		
		# Горизонтальный свайп → yaw
		yaw += delta_pos.x * 0.5
		# Вертикальный свайп → pitch
		pitch = clampf(pitch - delta_pos.y * 0.3, PITCH_MIN, PITCH_MAX)

# Колесо мыши для зума (для теста на компьютере)
func _unhandled_input(event: InputEvent) -> void:
	if event is InputEventMouseButton:
		if event.button_index == MOUSE_BUTTON_WHEEL_UP:
			zoom_level = clampf(zoom_level + 0.1, ZOOM_MIN, ZOOM_MAX)
		elif event.button_index == MOUSE_BUTTON_WHEEL_DOWN:
			zoom_level = clampf(zoom_level - 0.1, ZOOM_MIN, ZOOM_MAX)

func _on_jump() -> void:

	var other = gosha if ch == vanya else vanya

func _lock() -> void:

	touch.enabled = false
	if vanya: vanya.mode = vanya.Mode.IDLE

		controlled = ch
	_locked = false
	touch.enabled = true
	_set_control(controlled)

	for l in lines:
```
