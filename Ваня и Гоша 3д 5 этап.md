## 📷 Обновлённый `main_3d.gd` с управлением камерой

Замени файл `scripts/main_3d.gd` на этот код. Я добавил:
- **Зум** (приближение/отдаление) через переменную `zoom_level`
- **Наклон камеры** через переменную `pitch` (угол от 20° до 80°)
- **Плавные переходы** через tween-анимации
- **Управление жестами** (pinch-to-zoom для iOS)

```gdscript
extends Node3D

const VanyaScene = preload("res://scenes/vanya.tscn")
const GoshaScene = preload("res://scenes/gosha.tscn")
const BirdScene = preload("res://scenes/bird.tscn")
const DialogueScript = preload("res://scripts/dialogue_box.gd")
const TouchScript = preload("res://scripts/touch_controls.gd")
const CritterScript = preload("res://scripts/critter_3d.gd")
const WorldScript = preload("res://scripts/world_3d.gd")

const NAMES = { "vanya": "Ваня", "gosha": "Гоша", "bird": "Птичка" }
const COLORS = {
	"vanya": Color(1.0, 0.5, 0.45),
	"gosha": Color(0.55, 0.9, 0.5),
	"bird": Color(0.55, 0.78, 1.0),
}

enum State {
	INTRO_WALK,
	BIRD_SCENE,
	REFUSAL_WALK,
	END,
}

var state: int = State.INTRO_WALK
var world
var vanya: CharacterBody3D
var gosha: CharacterBody3D
var bird
var box: CanvasLayer
var touch: Control
var camera: Camera3D
var dim: CanvasModulate

var controlled = null
var holding: bool = true
var _locked: bool = false

# ---------- НАСТРОЙКИ КАМЕРЫ ----------
const CAM_DISTANCE := 12.0  # Базовое расстояние от персонажа
const CAM_HEIGHT := 10.0    # Базовая высота камеры

# Текущие значения (можно менять динамически)
var zoom_level: float = 1.0      # 1.0 = обычный вид, 2.0 = близко, 0.5 = далеко
var pitch: float = 40.0          # Угол наклона (20° = низко, 80° = почти сверху)

# Ограничения
const ZOOM_MIN := 0.4
const ZOOM_MAX := 2.5
const PITCH_MIN := 20.0
const PITCH_MAX := 80.0

# Для жестов (pinch-to-zoom)
var _touch_points: Dictionary = {}
var _initial_pinch_distance: float = 0.0
var _initial_zoom: float = 1.0

const CAM_LOOK_AT_OFFSET := Vector3(0, 0.5, 0)

# Лесные жители (3D координаты)
const CRITTERS_3D = [
	["squirrel", Vector3(15, 0, 3)],
	["rabbit", Vector3(8, 0, 12)],
	["hedgehog", Vector3(17, 0, 11)],
	["deer", Vector3(22, 0, 1)],
	["frog", Vector3(25, 0, 15)],
	["squirrel", Vector3(33, 0, 14)],
]

func _ready() -> void:
	# 1. Тёплый летний свет
	dim = CanvasModulate.new()
	dim.color = Color(1.0, 0.97, 0.88)
	add_child(dim)

	# 2. Перспективная камера под углом
	camera = Camera3D.new()
	camera.projection = Camera3D.PROJECTION_PERSPECTIVE
	camera.fov = 55.0
	add_child(camera)
	camera.make_current()
	
	# Установить начальную позицию камеры
	_update_camera_position()

	# 3. Лес
	world = WorldScript.new()
	add_child(world)

	# 4. Персонажи
	vanya = VanyaScene.instantiate()
	vanya.position = Vector3(0, 0, 0)
	add_child(vanya)

	gosha = GoshaScene.instantiate()
	gosha.position = Vector3(-1.5, 0, 0.5)
	gosha.enable_lag = true
	add_child(gosha)

	# 5. Птичка
	bird = BirdScene.instantiate()
	add_child(bird)

	# 6. Настройка walkable зон
	var walk: Array[Rect2] = world.walkable_start()
	var slow: Array[Rect2] = world.slow_zones()
	vanya.walkable = walk
	vanya.slow_zones = slow
	gosha.walkable = walk
	gosha.slow_zones = slow

	# 7. Лесные жители
	var water: Array[Rect2] = [
		Rect2(26, 16, 2.5, 4.5),
		Rect2(42, 16, 8.0, 5.0),
	]
	var no_zones: Array[Rect2] = []
	for spec in CRITTERS_3D:
		var cr = CritterScript.new()
		cr.kind = spec[0]
		cr.global_position = spec[1]
		cr.home = spec[1]
		cr.heroes = [vanya, gosha]
		cr.avoid = no_zones if spec[0] == "frog" else water
		add_child(cr)

	# 8. UI
	box = DialogueScript.new()
	add_child(box)
	
	touch = TouchScript.new()
	add_child(touch)

	_set_control(vanya)
	
	await get_tree().create_timer(0.5).timeout
	_run_act()

func _process(delta: float) -> void:
	if controlled != null and not _locked:
		var input_2d = touch.vector
		if Input.is_action_pressed("ui_right"): input_2d.x += 1.0
		if Input.is_action_pressed("ui_left"): input_2d.x -= 1.0
		if Input.is_action_pressed("ui_down"): input_2d.y += 1.0
		if Input.is_action_pressed("ui_up"): input_2d.y -= 1.0
		
		if input_2d.length() > 0.1:
			var cam_right = camera.global_transform.basis.x
			cam_right.y = 0
			cam_right = cam_right.normalized()
			
			var cam_forward = -camera.global_transform.basis.z
			cam_forward.y = 0
			cam_forward = cam_forward.normalized()
			
			var move_dir = cam_right * input_2d.x - cam_forward * input_2d.y
			controlled.input_vector = move_dir.limit_length(1.0)
		else:
			controlled.input_vector = Vector2.ZERO

func _physics_process(delta: float) -> void:
	if controlled != null:
		_update_camera_position()

# ---------- ОБНОВЛЕНИЕ ПОЗИЦИИ КАМЕРЫ ----------
func _update_camera_position() -> void:
	if controlled == null:
		return
	
	# Вычисляем позицию камеры на основе zoom и pitch
	var distance = CAM_DISTANCE / zoom_level
	var height = CAM_HEIGHT / zoom_level
	
	# Преобразуем pitch в радианы
	var pitch_rad = deg_to_rad(pitch)
	
	# Камера находится позади и выше персонажа
	var offset = Vector3(0, height, distance)
	offset = offset.rotated(Vector3.RIGHT, pitch_rad - PI/2)  # Наклон
	
	var target_pos = controlled.global_position + offset
	camera.global_position = camera.global_position.lerp(target_pos, 1.0 - exp(-5.0 * delta))
	
	var look_target = controlled.global_position + CAM_LOOK_AT_OFFSET
	camera.look_at(look_target, Vector3.UP)

# ---------- ПУБЛИЧНЫЕ МЕТОДЫ ДЛЯ УПРАВЛЕНИЯ КАМЕРОЙ ----------

# Приблизить камеру (плавно)
func zoom_in(target_zoom: float = 1.5, duration: float = 0.5) -> void:
	var tw = create_tween()
	tw.tween_method(_set_zoom, zoom_level, clampf(target_zoom, ZOOM_MIN, ZOOM_MAX), duration)

# Отдалить камеру (плавно)
func zoom_out(target_zoom: float = 0.7, duration: float = 0.5) -> void:
	var tw = create_tween()
	tw.tween_method(_set_zoom, zoom_level, clampf(target_zoom, ZOOM_MIN, ZOOM_MAX), duration)

# Изменить наклон камеры (плавно)
func set_pitch(target_pitch: float, duration: float = 0.5) -> void:
	var tw = create_tween()
	tw.tween_method(_set_pitch, pitch, clampf(target_pitch, PITCH_MIN, PITCH_MAX), duration)

# Внутренние методы для tween
func _set_zoom(value: float) -> void:
	zoom_level = value

func _set_pitch(value: float) -> void:
	pitch = value

# ---------- УПРАВЛЕНИЕ ЖЕСТАМИ (для iOS) ----------
func _input(event: InputEvent) -> void:
	if _locked:
		return
	
	# Отслеживаем касания
	if event is InputEventScreenTouch:
		if event.pressed:
			_touch_points[event.index] = event.position
		else:
			_touch_points.erase(event.index)
		
		# Если касаний стало 2 — начинаем pinch
		if _touch_points.size() == 2:
			var points = _touch_points.values()
			_initial_pinch_distance = points[0].distance_to(points[1])
			_initial_zoom = zoom_level
	
	# Pinch-to-zoom (два пальца)
	elif event is InputEventScreenDrag:
		if _touch_points.size() == 2 and _touch_points.has(event.index):
			_touch_points[event.index] = event.position
			var points = _touch_points.values()
			var current_distance = points[0].distance_to(points[1])
			
			if _initial_pinch_distance > 0:
				var zoom_delta = current_distance / _initial_pinch_distance
				zoom_level = clampf(_initial_zoom * zoom_delta, ZOOM_MIN, ZOOM_MAX)

# ---------- УПРАВЛЕНИЕ / ПЕРЕКЛЮЧЕНИЕ ----------
func _set_control(ch) -> void:
	controlled = ch
	var other = gosha if ch == vanya else vanya
	ch.mode = ch.Mode.CONTROLLED
	other.mode = other.Mode.FOLLOW

func _lock() -> void:
	_locked = true
	touch.enabled = false
	if vanya: vanya.mode = vanya.Mode.IDLE
	if gosha: gosha.mode = gosha.Mode.IDLE
	if controlled: controlled.input_vector = Vector2.ZERO

func _unlock(ch = null) -> void:
	if ch != null:
		controlled = ch
	_locked = false
	touch.enabled = true
	_set_control(controlled)

func _say(who: String, text: String) -> void:
	await box.say(NAMES[who], text, COLORS[who])

func _talk(lines: Array) -> void:
	_lock()
	for l in lines:
		await _say(l[0], l[1])
	box.hide_box()
	_unlock()

func _wait_until(cond: Callable) -> void:
	while not cond.call():
		await get_tree().process_frame

# ---------- СЦЕНАРИЙ АКТА (с птичкой) ----------
func _run_act() -> void:
	state = State.INTRO_WALK
	box.toast("Веди Ваню: води пальцем по левой половине экрана", Color.WHITE, 3.5)
	
	await _wait_until(func(): return vanya.global_position.z < -8.0)
	await _talk([["vanya", "Не отставай. До дома недалеко."]])
	
	await _wait_until(func(): return vanya.global_position.z < -16.0)
	await _talk([
		["gosha", "Там белка была…"],
		["vanya", "Потом. Сейчас главное — не задерживаться."],
	])

	# ВСТРЕЧА С СИНЕЙ ПТИЧКОЙ
	state = State.BIRD_SCENE
	await _wait_until(func(): return vanya.global_position.z < -24.0)
	_lock()
	gosha.enable_lag = false
	
	# Приближаем камеру для драматичного момента
	zoom_in(1.3, 1.0)
	
	# Птичка появляется справа и подлетает к Ване
	bird.global_position = Vector3(vanya.global_position.x + 6, 2.5, vanya.global_position.z - 2)
	await bird.fly_to(Vector3(vanya.global_position.x + 1.5, 1.2, vanya.global_position.z), 1.6)
	
	await _say("bird", "Подож… подождите.")
	await _say("bird", "Деду Морозу… очень плохо.")
	await _say("bird", "Правда плохо. Лесу нужна… ваша помощь.")
	await _say("bird", "Вы же… дети. Дети иногда могут то, что взрослые… не могут.")
	await _say("gosha", "Правда? А что случилось?")
	box.hide_box()
	
	# Возвращаем обычный зум
	zoom_out(1.0, 0.8)
	
	# Первое принудительное переключение: 6 секунд за Гошу
	await _tug_scene()
	
	await _say("vanya", "Мы идём домой.")
	await _say("vanya", "Нас не просили. И мы никому ничего не должны.")
	await _say("vanya", "Я отвечаю за нас двоих. Мы не будем ввязываться в чужие дела.")
	box.hide_box()
	
	# Птичка улетает
	bird.leave()
	
	_unlock(vanya)
	
	state = State.END
	await box.end_card("Конец Первого акта (3D Тест)", "Нажми, чтобы продолжить")
	get_tree().change_scene_to_file("res://scenes/menu.tscn")

# Гоша тянется к птичке, Ваня оттягивает его за руку (6 секунд)
func _tug_scene() -> void:
	box.flash()
	box.toast("Теперь ты — Гоша", COLORS["gosha"], 2.0)
	_locked = false
	touch.enabled = true
	controlled = gosha
	gosha.mode = gosha.Mode.CONTROLLED
	vanya.mode = vanya.Mode.IDLE
	
	var t: float = 0.0
	while t < 6.0:
		t += get_process_delta_time()
		var to_v: Vector3 = vanya.global_position - gosha.global_position
		var pull: float = 3.0 * minf(t / 3.0, 1.0)
		if to_v.length() < 0.8:
			pull = 0.0
		gosha.extra_pull = to_v.normalized() * pull
		await get_tree().process_frame
	
	gosha.extra_pull = Vector3.ZERO
	_lock()
	box.flash()
```

## 🎮 Что добавлено

### 1. **Управление зумом**
- `zoom_in(target_zoom, duration)` — плавно приблизить камеру
- `zoom_out(target_zoom, duration)` — плавно отдалить камеру
- Пример: `zoom_in(1.5, 1.0)` — приблизить в 1.5 раза за 1 секунду

### 2. **Управление наклоном**
- `set_pitch(target_pitch, duration)` — изменить угол наклона камеры
- Пример: `set_pitch(70.0, 1.0)` — наклонить камеру почти сверху за 1 секунду

### 3. **Жесты для iOS (pinch-to-zoom)**
- Два пальца на экране → приближение/отдаление
- Работает автоматически, не нужно ничего вызывать

### 4. **Автоматический зум в сценарии**
- Когда появляется птичка — камера плавно приближается (`zoom_in(1.3, 1.0)`)
- После диалога — возвращается к обычному виду (`zoom_out(1.0, 0.8)`)
